##### RERUNNING THE FAILED SAMPLES 

# Grab just the six that didn't zip for whatever reason 
ls fastqs/*.fastq | xargs -n1 basename -s .fastq > SRA_LIST_V1failed.txt
 - list all files in the fastqs directory that has something.fastq in the file name 
 - pipe that list into xargs to run through each item of the list individually (-n1) 
 - basename strips the directory information 
 - s flag strips the suffix (after deliminator in this case = .fastq)
 - Save all of that in a text file named SRA_LIST_V1failed. txt 

# Strip off the _# suffix to remove both paired ends (_1; _2)
	# Remove underscore and either a one or a two at the end of the name ($); replace with nothing (//) and save to tmp file
sed 's/_[12]$//' SRA_LIST_V1failed.txt > tmp.txt
	# Rename 
mv tmp.txt SRA_LIST_V1failed.txt 
	# Check 
cat SRA_LIST_V1failed.txt
 
# Remove both paired end samples from folders to restart (pipeline will see it as already done otherwise)
	# This just prints the files that will be removed for a given folder
xargs -a SRA_LIST_V1failed.txt -I {} rm -f fastqs/{}_1.fastq.gz fastqs/{}_2.fastq
	- where ' -I {}' will index the contents of xargs and put each item in a {placeholder}
	- f flag prevents errors if one or other is missing; might want to remove that
	- remove 'echo' to actually run the command
